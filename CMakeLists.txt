# Release ビルドはデバッグシンボルを削除し最適化を有効化する
# （仕様書 3. クライアント: 「Release構成時のデバッグシンボル削除、コード最適化」）
if (MSVC)
    # ソースファイルは UTF-8（日本語コメントを含む）。これを明示しないと
    # 日本語環境(CP932)ではコメント末尾が次行を巻き込み #define が壊れる。
    # （warning C4819 と "undeclared identifier" の根本原因）
    add_compile_options(/utf-8)
    add_compile_options($<$<CONFIG:Release>:/O2>)
    add_link_options($<$<CONFIG:Release>:/DEBUG:NONE>)
    add_link_options($<$<CONFIG:Release>:/INCREMENTAL:NO>)
    if(PROTECT_DISTRIBUTION)
        include(CheckIPOSupported)
        check_ipo_supported(RESULT _ipo_ok OUTPUT _ipo_error)
        if(NOT _ipo_ok)
            message(FATAL_ERROR "Protected Release requires IPO: ${_ipo_error}")
        endif()
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
        add_compile_definitions($<$<CONFIG:Release>:PROTECTED_DISTRIBUTION=1>)
        add_compile_options($<$<CONFIG:Release>:/Gy> $<$<CONFIG:Release>:/Gw> $<$<CONFIG:Release>:/GF> $<$<CONFIG:Release>:/guard:cf>)
        add_link_options($<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF> $<$<CONFIG:Release>:/guard:cf>
            $<$<CONFIG:Release>:/DYNAMICBASE> $<$<CONFIG:Release>:/HIGHENTROPYVA> $<$<CONFIG:Release>:/NXCOMPAT>)
    endif()
endif()

# ---------------------------------------------------------------------------
# HLSL → DXIL コンパイル（dxc.exe）
#   出力先: ${CMAKE_BINARY_DIR}/shaders/*.dxil
#   dxc が見つからない場合は警告のみ（CMake 構成自体は通す）
# ---------------------------------------------------------------------------
find_program(DXC_EXE dxc
    HINTS
        "$ENV{WindowsSdkVerBinPath}x64"
        "$ENV{WindowsSdkDir}bin/x64"
    PATH_SUFFIXES x64)

# 上で見つからない場合: Windows SDK の bin/<version>/x64 を総当たりして
# 最新版の dxc.exe を選ぶ（SDK はバージョン番号フォルダを挟むため）。
if (NOT DXC_EXE)
    file(GLOB _dxc_candidates
        "C:/Program Files (x86)/Windows Kits/10/bin/*/x64/dxc.exe"
        "C:/Program Files/Windows Kits/10/bin/*/x64/dxc.exe"
        "$ENV{ProgramFiles}/Windows Kits/10/bin/*/x64/dxc.exe")
    if (_dxc_candidates)
        list(SORT _dxc_candidates)
        list(GET _dxc_candidates -1 DXC_EXE) # 末尾 = 最新バージョン
        message(STATUS "dxc を SDK から検出: ${DXC_EXE}")
    endif()
endif()

set(SHADER_BIN_DIR ${CMAKE_BINARY_DIR}/shaders)
file(MAKE_DIRECTORY ${SHADER_BIN_DIR})

if (DXC_EXE)
    message(STATUS "dxc found: ${DXC_EXE}")

    function(compile_shader SRC ENTRY PROFILE OUT)
        set(_shader_strip)
        if(PROTECT_DISTRIBUTION)
            set(_shader_strip -Qstrip_debug -Qstrip_reflect)
        endif()
        add_custom_command(
            OUTPUT ${SHADER_BIN_DIR}/${OUT}
            COMMAND ${DXC_EXE} -E ${ENTRY} -T ${PROFILE}
                    ${_shader_strip}
                    -Fo ${SHADER_BIN_DIR}/${OUT}
                    ${CMAKE_SOURCE_DIR}/assets/shaders/${SRC}
            DEPENDS ${CMAKE_SOURCE_DIR}/assets/shaders/${SRC}
            COMMENT "dxc compiling ${SRC} -> ${OUT}"
            VERBATIM)
    endfunction()

    compile_shader(SimpleVS.hlsl VSMain vs_6_0 SimpleVS.dxil)
    compile_shader(SimplePS.hlsl PSMain ps_6_0 SimplePS.dxil)
    compile_shader(OverlayVS.hlsl VSMain vs_6_0 OverlayVS.dxil)
    compile_shader(OverlayPS.hlsl PSMain ps_6_0 OverlayPS.dxil)
    if (NOT GAME_PROFILE STREQUAL "range")
    compile_shader(TexturedVS.hlsl VSMain vs_6_0 TexturedVS.dxil)
    compile_shader(TexturedPS.hlsl PSMain ps_6_0 TexturedPS.dxil)
    # ステージL-03b: シャドウパス用（深度専用なので VS のみ。PS は無い）
    compile_shader(ShadowVS.hlsl VSMain vs_6_0 ShadowVS.dxil)
    # ステージL-02: PBR（Cook-Torrance + 法線マップ）
    compile_shader(PbrVS.hlsl VSMain vs_6_0 PbrVS.dxil)
    compile_shader(PbrPS.hlsl PSMain ps_6_0 PbrPS.dxil)
    # ステージL-04b: GPU スキニング
    compile_shader(PbrSkinnedVS.hlsl VSMain vs_6_0 PbrSkinnedVS.dxil)
    compile_shader(ShadowSkinnedVS.hlsl VSMain vs_6_0 ShadowSkinnedVS.dxil)
    endif()

    if (GAME_PROFILE STREQUAL "range")
    add_custom_target(shaders ALL DEPENDS
        ${SHADER_BIN_DIR}/SimpleVS.dxil ${SHADER_BIN_DIR}/SimplePS.dxil
        ${SHADER_BIN_DIR}/OverlayVS.dxil ${SHADER_BIN_DIR}/OverlayPS.dxil)
    else()
    add_custom_target(shaders ALL
        DEPENDS
            ${SHADER_BIN_DIR}/SimpleVS.dxil
            ${SHADER_BIN_DIR}/SimplePS.dxil
            ${SHADER_BIN_DIR}/OverlayVS.dxil
            ${SHADER_BIN_DIR}/OverlayPS.dxil
            ${SHADER_BIN_DIR}/TexturedVS.dxil
            ${SHADER_BIN_DIR}/TexturedPS.dxil
            ${SHADER_BIN_DIR}/ShadowVS.dxil
            ${SHADER_BIN_DIR}/PbrVS.dxil
            ${SHADER_BIN_DIR}/PbrPS.dxil
            ${SHADER_BIN_DIR}/PbrSkinnedVS.dxil
            ${SHADER_BIN_DIR}/ShadowSkinnedVS.dxil)
    endif()
    add_dependencies(game shaders)
else()
    message(WARNING
        "dxc.exe が見つかりません。Windows SDK の bin/x64 を PATH に追加するか、"
        "手動で assets/shaders/*.hlsl を ${SHADER_BIN_DIR} にコンパイルしてください。")
endif()

# ---------------------------------------------------------------------------
# パッケージ化（ステージJ-05）: 通常版(normal) / 対策版(hardened) を配布フォルダ・zip に。
#
#   -DHARDENED=ON でビルドすると HARDENED_BUILD を定義し、ゲームは防御（J-03 深度拒絶＋
#   J-04 タグマスク＋難読化）を「既定 ON」で起動する（normal は既定 OFF）。
#   ゲーム本体のコード・起動コマンドは共通。既定値だけがビルドで変わる（ゲーム内 P/M/L1/R1 で
#   実行時に切替も可能）。①のフック型チート(cheat_tester --wallhack)は normal では成立し、
#   hardened では門番＋マスクで不成立になる、をパッケージ単位で検証できる。
#
#   ビルド例:
#     cmake -B build-normal   -A x64 -DHARDENED=OFF && cmake --build build-normal   --config Release
#     cmake -B build-hardened -A x64 -DHARDENED=ON  && cmake --build build-hardened --config Release
#   パッケージ生成:
#     cd build-normal   && cpack -C Release   → GameEngine-normal.zip
#     cd build-hardened && cpack -C Release   → GameEngine-hardened.zip
# ---------------------------------------------------------------------------
option(HARDENED "アンチチート防御を既定 ON にした対策版パッケージをビルドする" OFF)
if (GAME_PROFILE STREQUAL "range")
    if (HARDENED)
        message(FATAL_ERROR "range has no security module; HARDENED is only valid for all/fps")
    endif()
    message(STATUS "RANGE build: offline common game; no local defense services")
elseif (HARDENED)
    target_compile_definitions(game PRIVATE HARDENED_BUILD=1)
    message(STATUS "HARDENED build: 防御（J-03/J-04/難読化）を既定 ON でパッケージします。")
else()
    message(STATUS "NORMAL build: fps/hide defenses default OFF; offline card defaults shuffle+watchdog.")
endif()
