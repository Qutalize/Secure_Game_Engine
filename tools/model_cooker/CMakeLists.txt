cmake_minimum_required(VERSION 3.24)
project(FbxCooker LANGUAGES C CXX)
add_executable(fbx_cooker main.cpp ../../third_party/ufbx/ufbx.c)
target_compile_features(fbx_cooker PRIVATE cxx_std_20)
target_include_directories(fbx_cooker PRIVATE ../../third_party/ufbx)
target_compile_definitions(fbx_cooker PRIVATE UFBX_NO_SUBDIVISION UFBX_NO_GEOMETRY_CACHE UFBX_NO_OBJ)
if(MSVC)
 target_compile_options(fbx_cooker PRIVATE /utf-8)
endif()
